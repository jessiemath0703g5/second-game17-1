        // --- Game Logic ---
        function startGame() {
            if (!player.levelId || !player.diffTime) { alert("請選擇關卡與難度！"); return; }
            gameState = { 
                score: 0, timeLeft: player.diffTime, combo: 0, isPaused: false, 
                totalQuestions: 0, correctQuestions: 0, isShowingFeedback: false,
                history: []
            };
            currentShuffled = []; // Reset shuffle deck for new game
            
            hideFeedback();
            
            updateStatsUI();
            switchScreen('screen-game');
            nextQuestion();
            
            if (gameState.timerId) clearInterval(gameState.timerId);
            gameState.timerId = setInterval(timerTick, 1000);
            
            initWhiteboard();
        }

        function restartGame() {
            if (confirm("確定要重新開始嗎？這將會重置目前的分數與時間。")) {
                startGame();
            }
        }

        function timerTick() {
            if (gameState.isPaused) return;
            gameState.timeLeft--;
            document.getElementById('timerDisplay').innerText = gameState.timeLeft;
            if (gameState.timeLeft <= 0) {
                timeUpEndGame();
            }
        }

        function pauseGame() {
            gameState.isPaused = !gameState.isPaused;
            document.getElementById('pauseBtn').innerText = gameState.isPaused ? "▶️ 繼續" : "⏸️ 暫停";
        }

        function nextQuestion() {
            let q = generateQuestion();
            document.getElementById('questionText').innerText = q.text;
            document.getElementById('questionImageContainer').innerHTML = q.svg;
            gameState.currentAns = q.answer;
            let inp = document.getElementById('answerInput');
            inp.value = "";
            inp.disabled = false;
            document.getElementById('submitBtn').disabled = false;
            inp.focus();
        }

        function handleEnter(e) { if (e.key === 'Enter') checkAnswer(); }

        function checkAnswer() {
            if (gameState.isPaused || gameState.isShowingFeedback) return;
            let userAnsStr = document.getElementById('answerInput').value.trim();
            if (userAnsStr === "") return;
            
            gameState.isShowingFeedback = true;
            document.getElementById('answerInput').disabled = true;
            document.getElementById('submitBtn').disabled = true;

            gameState.totalQuestions++;
            let parsedUser = parseFloat(userAnsStr);
            // Allow precision margin of 0.01 for correctness
            let isCorrect = Math.abs(parsedUser - gameState.currentAns) <= 0.015;

            // Record History
            gameState.history.push({
                text: gameState.currentQuestionText,
                userAns: userAnsStr,
                correctAns: gameState.currentAns,
                isCorrect: isCorrect
            });

            if (isCorrect) {
                gameState.correctQuestions++;
                gameState.combo++;
                gameState.timeLeft += 5; // Reward
                let points = Math.floor(10 * (1 + (gameState.combo - 1) * 0.2));
                gameState.score += points;
                showFeedback(true, gameState.currentAns);
            } else {
                gameState.combo = 0;
                gameState.timeLeft = Math.max(0, gameState.timeLeft - 5); // Penalty
                showFeedback(false, gameState.currentAns);
            }
            
            updateStatsUI();
            
            // Wait 1.5s, then hide feedback and go next
            setTimeout(() => {
                hideFeedback();
                if (gameState.timeLeft > 0) {
                    nextQuestion();
                } else {
                    timeUpEndGame();
                }
                gameState.isShowingFeedback = false;
            }, 1500);
        }

        // --- Whiteboard Logic ---
        let canvas, ctx;
        let isDrawing = false;
        let isWbActive = false;

        // ✅ 修正：精準取得畫布與滑鼠/手指之間的相對座標
        function getWbPos(e) {
            let rect = canvas.getBoundingClientRect();
            if (e.touches && e.touches.length > 0) {
                return { 
                    x: e.touches[0].clientX - rect.left, 
                    y: e.touches[0].clientY - rect.top 
                };
            }
            return { 
                x: e.clientX - rect.left, 
                y: e.clientY - rect.top 
            };
        }

        function initWhiteboard() {
            canvas = document.getElementById('whiteboard');
            ctx = canvas.getContext('2d');
            let container = document.getElementById('whiteboard-container');
            canvas.width = container.offsetWidth;
            canvas.height = container.offsetHeight;
            
            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            ctx.strokeStyle = '#e74c3c';

            canvas.onmousedown = (e) => { 
                isDrawing = true; 
                let pos = getWbPos(e);
                ctx.beginPath(); 
                ctx.moveTo(pos.x, pos.y); 
            };
            canvas.onmousemove = (e) => { 
                if(isDrawing) { 
                    let pos = getWbPos(e);
                    ctx.lineTo(pos.x, pos.y); 
                    ctx.stroke(); 
                } 
            };
            canvas.onmouseup = () => isDrawing = false;
            canvas.onmouseout = () => isDrawing = false;
            
            // Touch support
            canvas.ontouchstart = (e) => { 
                e.preventDefault(); 
                isDrawing = true; 
                let pos = getWbPos(e);
                ctx.beginPath(); 
                ctx.moveTo(pos.x, pos.y); 
            };
            canvas.ontouchmove = (e) => { 
                if(isDrawing) { 
                    e.preventDefault();
                    let pos = getWbPos(e);
                    ctx.lineTo(pos.x, pos.y); 
                    ctx.stroke(); 
                } 
            };
            canvas.ontouchend = () => isDrawing = false;
        }

        function toggleWhiteboard() {
            let gameScreen = document.getElementById('screen-game');

            isWbActive = !isWbActive;
            if(isWbActive) {

                btn.style.background = "#e67e22";

                gameScreen.classList.remove('wb-active');

            }
        }

        function clearWhiteboard() {
            if(ctx && canvas) ctx.clearRect(0, 0, canvas.width, canvas.height);
        }

        function saveToLeaderboard() {
            if (gameState.totalQuestions === 0) return; // don't save 0 questions played
            let record = {
                name: player.name,
                level: player.levelName,
                diff: player.diffName,
                correct: gameState.correctQuestions,
                total: gameState.totalQuestions,
                score: gameState.score,
                date: new Date().toLocaleString('zh-TW', {hour12:false})
            };
            let lb = JSON.parse(localStorage.getItem('jessieMathLB_G5_v5')) || [];
            lb.push(record);
            lb.sort((a, b) => b.score - a.score); // 總分由高到低
            lb = lb.slice(0, 10); // 取前10名
            localStorage.setItem('jessieMathLB_G5_v5', JSON.stringify(lb));
        }

        function renderLeaderboard() {
            let lb = JSON.parse(localStorage.getItem('jessieMathLB_G5_v5')) || [];
            let tbody = document.getElementById('leaderboardBody');
            tbody.innerHTML = "";
            lb.forEach((r, idx) => {
                let rankIcon = `${idx + 1}`;
                if (idx === 0) rankIcon = "🥇";
                if (idx === 1) rankIcon = "🥈";
                if (idx === 2) rankIcon = "🥉";
                if (idx > 2 && idx < 10) rankIcon = "🏅";
                
                let tr = document.createElement('tr');
                tr.innerHTML = `
                    <td class="rank-icon">${rankIcon}</td>
                    <td>${r.name}</td>
                    <td>${r.level}</td>
                    <td>${r.diff}</td>
                    <td>${r.correct} / ${r.total}</td>
                    <td><b>${r.score}</b></td>
                    <td style="font-size:13px; color:#7f8c8d;">${r.date}</td>
                `;
                tbody.appendChild(tr);
            });
            if (lb.length === 0) {
                tbody.innerHTML = "<tr><td colspan='7'>目前尚無紀錄，快來成為第一名吧！</td></tr>";
            }
        }
